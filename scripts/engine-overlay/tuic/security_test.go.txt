func securityInput(t *testing.T, cert, key string) darkproadapter.Input {
	return inputFor(t, "cubic", dt.Port(t), cert, key, []tuic.TuicClientSettings{{UUID: "11111111-1111-4111-8111-111111111111", Password: "secret-for-security-tests", Email: "alice"}})
}
func mutateTLS(t *testing.T, in darkproadapter.Input, fn func(*config)) darkproadapter.Input {
	t.Helper()
	var cfg config
	if e := json.Unmarshal(in.Config, &cfg); e != nil {
		t.Fatal(e)
	}
	fn(&cfg)
	in.Config, _ = json.Marshal(cfg)
	return in
}
func handshake(t *testing.T, in darkproadapter.Input, client *tls.Config) (*quic.Conn, error) {
	t.Helper()
	var cfg config
	_ = json.Unmarshal(in.Config, &cfg)
	ctx, cancel := context.WithTimeout(t.Context(), 3*time.Second)
	defer cancel()
	c, e := quic.DialAddr(ctx, cfg.Listen, client, &quic.Config{EnableDatagrams: true})
	if e == nil {
		t.Cleanup(func() { _ = c.CloseWithError(0, "") })
	}
	return c, e
}
func trust(roots *x509.CertPool, sni string) *tls.Config {
	return &tls.Config{RootCAs: roots, ServerName: sni, NextProtos: []string{"h3"}, MinVersion: tls.VersionTLS13}
}

func TestTUICRealECHAndSNIRejection(t *testing.T) {
	cert, key, roots := testCertificate(t)
	private, e := ecdh.X25519().GenerateKey(rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	body := []byte{7, 0, 0x20}
	body = binary.BigEndian.AppendUint16(body, uint16(len(private.PublicKey().Bytes())))
	body = append(body, private.PublicKey().Bytes()...)
	body = append(body, 0, 4, 0, 1, 0, 1, 0)
	publicName := "public.example"
	body = append(body, byte(len(publicName)))
	body = append(body, []byte(publicName)...)
	body = append(body, 0, 0)
	configBytes := binary.BigEndian.AppendUint16([]byte{0xfe, 0x0d}, uint16(len(body)))
	configBytes = append(configBytes, body...)
	serverKeys := binary.BigEndian.AppendUint16(nil, uint16(len(private.Bytes())))
	serverKeys = append(serverKeys, private.Bytes()...)
	serverKeys = binary.BigEndian.AppendUint16(serverKeys, uint16(len(configBytes)))
	serverKeys = append(serverKeys, configBytes...)
	clientConfig := binary.BigEndian.AppendUint16(nil, uint16(len(configBytes)))
	clientConfig = append(clientConfig, configBytes...)
	in := mutateTLS(t, securityInput(t, cert, key), func(cfg *config) {
		cfg.TLS["echServerKeys"], _ = json.Marshal(base64.StdEncoding.EncodeToString(serverKeys))
	})
	managed(t, in)
	clientTLS := trust(roots, "localhost")
	clientTLS.EncryptedClientHelloConfigList = clientConfig
	c, e := handshake(t, in, clientTLS)
	if e != nil {
		t.Fatalf("real ECH QUIC handshake: %v", e)
	}
	if !c.ConnectionState().TLS.ECHAccepted {
		t.Fatal("ECH was silently ignored")
	}
	rejected := trust(roots, "unrecognized.example")
	rejected.InsecureSkipVerify = true
	if c, e := handshake(t, in, rejected); e == nil {
		c.CloseWithError(0, "")
		t.Fatal("unknown SNI admitted")
	}
	serverKeys[2] ^= 0xff
	broken := mutateTLS(t, in, func(cfg *config) {
		cfg.TLS["echServerKeys"], _ = json.Marshal(base64.StdEncoding.EncodeToString(serverKeys))
	})
	if _, e := prepare(broken); e == nil {
		t.Fatal("mismatched ECH private key accepted")
	}
}
func TestTUICCertificateSnapshotAndPrivateKeyLog(t *testing.T) {
	cert, key, roots := testCertificate(t)
	dir := t.TempDir()
	certFile, keyFile, logFile := filepath.Join(dir, "certificate.pem"), filepath.Join(dir, "key.pem"), filepath.Join(dir, "tls.keys")
	if e := os.WriteFile(certFile, []byte(cert), 0600); e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(keyFile, []byte(key), 0600); e != nil {
		t.Fatal(e)
	}
	in := mutateTLS(t, securityInput(t, cert, key), func(cfg *config) {
		cfg.TLS["certificates"], _ = json.Marshal([]map[string]any{{"certificateFile": certFile, "keyFile": keyFile}})
		cfg.TLS["masterKeyLog"], _ = json.Marshal(logFile)
		cfg.TLS["curvePreferences"] = json.RawMessage(`["X25519"]`)
		cfg.TLS["enableSessionResumption"] = json.RawMessage("true")
	})
	plan, e := prepare(in)
	if e != nil {
		t.Fatal(e)
	}
	// A generation must not reread mutable file references after validation.
	if e = os.WriteFile(certFile, []byte("replaced after prepare"), 0600); e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(keyFile, []byte("replaced after prepare"), 0600); e != nil {
		t.Fatal(e)
	}
	w := dt.Worker(t, in, plan)
	dt.Command(t, w, "activate", dt.ActivateNonce)
	c, e := handshake(t, in, trust(roots, "localhost"))
	if e != nil {
		t.Fatalf("frozen certificate handshake: %v", e)
	}
	if c.ConnectionState().TLS.Version != tls.VersionTLS13 || c.ConnectionState().TLS.NegotiatedProtocol != "h3" {
		t.Fatal("TLS/ALPN request not applied")
	}
	raw, e := os.ReadFile(logFile)
	if e != nil || len(raw) == 0 {
		t.Fatal("explicit TLS key log did not record handshake")
	}
	info, e := os.Stat(logFile)
	if e != nil || info.Mode().Perm() != 0600 {
		t.Fatal("TLS key log is not private")
	}
	dt.Final(t, w)
}
func caFixture(t *testing.T) (*x509.Certificate, *ecdsa.PrivateKey, string, string) {
	t.Helper()
	key, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	tmpl := &x509.Certificate{SerialNumber: big.NewInt(40), Subject: pkix.Name{CommonName: "local acceptance CA"}, IsCA: true, BasicConstraintsValid: true, KeyUsage: x509.KeyUsageCertSign | x509.KeyUsageCRLSign, NotBefore: time.Now().Add(-time.Hour), NotAfter: time.Now().Add(time.Hour)}
	raw, e := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if e != nil {
		t.Fatal(e)
	}
	cert, e := x509.ParseCertificate(raw)
	if e != nil {
		t.Fatal(e)
	}
	private, e := x509.MarshalPKCS8PrivateKey(key)
	if e != nil {
		t.Fatal(e)
	}
	return cert, key, string(pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: raw})), string(pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: private}))
}
func TestTUICRealOCSPStaplingAndInvalidResponse(t *testing.T) {
	ca, caKey, caPEM, _ := caFixture(t)
	var answer atomic.Value
	answer.Store([]byte("not ready"))
	responder := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Method != "POST" || r.Header.Get("Content-Type") != "application/ocsp-request" {
			http.Error(w, "bad OCSP request", 400)
			return
		}
		w.Header().Set("Content-Type", "application/ocsp-response")
		_, _ = w.Write(answer.Load().([]byte))
	}))
	defer responder.Close()
	key, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	tmpl := &x509.Certificate{SerialNumber: big.NewInt(41), Subject: pkix.Name{CommonName: "localhost"}, DNSNames: []string{"localhost"}, NotBefore: time.Now().Add(-time.Hour), NotAfter: time.Now().Add(time.Hour), KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth}, OCSPServer: []string{responder.URL}}
	der, e := x509.CreateCertificate(rand.Reader, tmpl, ca, &key.PublicKey, caKey)
	if e != nil {
		t.Fatal(e)
	}
	leaf, e := x509.ParseCertificate(der)
	if e != nil {
		t.Fatal(e)
	}
	private, e := x509.MarshalPKCS8PrivateKey(key)
	if e != nil {
		t.Fatal(e)
	}
	cert := string(pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der})) + caPEM
	keyPEM := string(pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: private}))
	response, e := ocsp.CreateResponse(ca, ca, ocsp.Response{Status: ocsp.Good, SerialNumber: leaf.SerialNumber, ThisUpdate: time.Now().Add(-time.Minute), NextUpdate: time.Now().Add(time.Hour)}, caKey)
	if e != nil {
		t.Fatal(e)
	}
	answer.Store(response)
	roots := x509.NewCertPool()
	roots.AddCert(ca)
	in := mutateTLS(t, securityInput(t, cert, keyPEM), func(cfg *config) {
		cfg.TLS["certificates"], _ = json.Marshal([]map[string]any{{"certificate": strings.Split(cert, "\n"), "key": strings.Split(keyPEM, "\n"), "ocspStapling": 60}})
	})
	w := managed(t, in)
	c, e := handshake(t, in, trust(roots, "localhost"))
	if e != nil {
		t.Fatal(e)
	}
	if !bytes.Equal(c.ConnectionState().TLS.OCSPResponse, response) {
		t.Fatal("requested OCSP staple absent from actual QUIC handshake")
	}
	dt.Final(t, w)
	answer.Store([]byte("invalid signed response"))
	in = mutateTLS(t, in, func(cfg *config) { cfg.Listen = net.JoinHostPort("127.0.0.1", strconv.Itoa(dt.Port(t))) })
	managed(t, in)
	if c, e := handshake(t, in, trust(roots, "localhost")); e == nil {
		c.CloseWithError(0, "")
		t.Fatal("invalid OCSP response was silently ignored")
	}
}
func TestTUICRealCertificateIssuanceAndChain(t *testing.T) {
	ca, _, caPEM, keyPEM := caFixture(t)
	roots := x509.NewCertPool()
	roots.AddCert(ca)
	in := mutateTLS(t, securityInput(t, caPEM, keyPEM), func(cfg *config) {
		cfg.TLS["certificates"], _ = json.Marshal([]map[string]any{{"certificate": strings.Split(caPEM, "\n"), "key": strings.Split(keyPEM, "\n"), "usage": "issue", "buildChain": true}})
	})
	managed(t, in)
	c, e := handshake(t, in, trust(roots, "issued.example"))
	if e != nil {
		t.Fatal(e)
	}
	chain := c.ConnectionState().TLS.PeerCertificates
	if len(chain) < 2 || chain[0].VerifyHostname("issued.example") != nil {
		t.Fatal("issuer/build-chain option was not applied")
	}
}
