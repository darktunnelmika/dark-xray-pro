type certificateInput struct {
	ID              string   `json:"id"`
	Certificate     []string `json:"certificate"`
	Key             []string `json:"key"`
	CertificateFile string   `json:"certificateFile"`
	KeyFile         string   `json:"keyFile"`
	Usage           string   `json:"usage"`
	OCSP            int      `json:"ocspStapling"`
	Once            bool     `json:"oneTimeLoading"`
	BuildChain      bool     `json:"buildChain"`
}
type servingCertificate struct {
	pair       tls.Certificate
	usage      string
	buildChain bool
	ocspEvery  time.Duration
	mu         sync.Mutex
	staple     []byte
	refresh    time.Time
}

func tlsFor(cfg config) (*tls.Config, string, string, error) {
	fail := func() (*tls.Config, string, string, error) {
		return nil, "", "", errors.New("invalid or incompatible TUIC TLS option")
	}
	supported := map[string]bool{"certificates": true, "serverName": true, "minVersion": true, "maxVersion": true, "alpn": true, "rejectUnknownSni": true, "enableSessionResumption": true, "curvePreferences": true, "disableSystemRoot": true, "useSystemCA": true, "cipherSuites": true, "echServerKeys": true, "masterKeyLog": true}
	for k := range cfg.TLS {
		if !supported[k] {
			return fail()
		}
	}
	out := &tls.Config{MinVersion: tls.VersionTLS13, MaxVersion: tls.VersionTLS13, SessionTicketsDisabled: true}
	stringValue := func(k string) (string, error) {
		raw := cfg.TLS[k]
		if len(raw) == 0 {
			return "", nil
		}
		var v string
		e := json.Unmarshal(raw, &v)
		return v, e
	}
	boolValue := func(k string) (bool, error) {
		raw := cfg.TLS[k]
		if len(raw) == 0 {
			return false, nil
		}
		var v bool
		e := json.Unmarshal(raw, &v)
		return v, e
	}
	for _, k := range []string{"minVersion", "maxVersion"} {
		s, e := stringValue(k)
		if e != nil {
			return fail()
		}
		if k == "maxVersion" && s != "" && s != "1.3" {
			return nil, "", "", errors.New("TUIC QUIC requires TLS 1.3")
		}
		if s != "" && s != "1.0" && s != "1.1" && s != "1.2" && s != "1.3" {
			return fail()
		}
	}
	if s, e := stringValue("cipherSuites"); e != nil {
		return fail()
	} else if s != "" {
		return nil, "", "", errors.New("TUIC TLS 1.3 cipher suites cannot be overridden by TLS 1.2 settings")
	}
	if s, e := stringValue("serverName"); e != nil {
		return fail()
	} else {
		out.ServerName = s
	}
	disabled, e := boolValue("disableSystemRoot")
	if e != nil {
		return fail()
	}
	system, e := boolValue("useSystemCA")
	if e != nil {
		return fail()
	}
	if !disabled || system {
		out.RootCAs, e = x509.SystemCertPool()
		if e != nil {
			return fail()
		}
	} else {
		out.RootCAs = x509.NewCertPool()
	}
	var certs []certificateInput
	if raw := cfg.TLS["certificates"]; len(raw) > 0 {
		if darkproadapter.Decode(raw, &certs) != nil {
			return fail()
		}
	}
	if cfg.Server.Certificate != "" || cfg.Server.PrivateKey != "" {
		certs = append([]certificateInput{{CertificateFile: cfg.Server.Certificate, KeyFile: cfg.Server.PrivateKey}}, certs...)
	}
	firstCertificate, firstKey := "", ""
	var serving []*servingCertificate
	var issuing []*servingCertificate
	for _, c := range certs {
		if c.OCSP < 0 || c.OCSP > 31536000 {
			return fail()
		}
		certificate := strings.Join(c.Certificate, "\n")
		key := strings.Join(c.Key, "\n")
		if certificate == "" {
			certificate = c.CertificateFile
		}
		if key == "" {
			key = c.KeyFile
		}
		certificate, e = darkproadapter.Material(certificate)
		if e != nil {
			return nil, "", "", e
		}
		if c.Usage == "verify" {
			if !out.RootCAs.AppendCertsFromPEM([]byte(certificate)) {
				return fail()
			}
			continue
		}
		if c.Usage != "" && c.Usage != "encipherment" && c.Usage != "issue" {
			return fail()
		}
		key, e = darkproadapter.Material(key)
		if e != nil {
			return nil, "", "", e
		}
		pair, e := tls.X509KeyPair([]byte(certificate), []byte(key))
		if e != nil || len(pair.Certificate) == 0 {
			return fail()
		}
		pair.Leaf, e = x509.ParseCertificate(pair.Certificate[0])
		if e != nil {
			return fail()
		}
		one := &servingCertificate{pair: pair, usage: c.Usage, buildChain: c.BuildChain, ocspEvery: time.Duration(c.OCSP) * time.Second}
		if c.Usage == "issue" {
			if !pair.Leaf.IsCA || pair.Leaf.KeyUsage&x509.KeyUsageCertSign == 0 {
				return fail()
			}
			issuing = append(issuing, one)
		} else {
			serving = append(serving, one)
			out.Certificates = append(out.Certificates, pair)
		}
		if firstCertificate == "" {
			firstCertificate, firstKey = certificate, key
		}
	}
	if firstCertificate == "" {
		return fail()
	}
	out.NextProtos = append([]string(nil), cfg.Server.ALPN...)
	if len(out.NextProtos) == 0 {
		if v := cfg.TLS["alpn"]; len(v) > 0 {
			if json.Unmarshal(v, &out.NextProtos) != nil {
				return fail()
			}
		}
	}
	if len(out.NextProtos) == 0 {
		out.NextProtos = []string{"h3", "spdy/3.1"}
	}
	for _, p := range out.NextProtos {
		if len(p) == 0 || len(p) > 255 {
			return fail()
		}
	}
	if v := cfg.TLS["enableSessionResumption"]; len(v) > 0 {
		var enabled bool
		if json.Unmarshal(v, &enabled) != nil {
			return fail()
		}
		out.SessionTicketsDisabled = !enabled
	}
	if v := cfg.TLS["curvePreferences"]; len(v) > 0 {
		var values []string
		if json.Unmarshal(v, &values) != nil {
			return fail()
		}
		for _, name := range values {
			curve, ok := map[string]tls.CurveID{"CurveP256": tls.CurveP256, "P256": tls.CurveP256, "CurveP384": tls.CurveP384, "P384": tls.CurveP384, "CurveP521": tls.CurveP521, "P521": tls.CurveP521, "X25519": tls.X25519, "X25519MLKEM768": tls.X25519MLKEM768}[name]
			if !ok {
				return fail()
			}
			out.CurvePreferences = append(out.CurvePreferences, curve)
		}
	}
	if encoded, e := stringValue("echServerKeys"); e != nil {
		return fail()
	} else if encoded != "" {
		raw, e := base64.StdEncoding.DecodeString(encoded)
		if e != nil {
			return fail()
		}
		out.EncryptedClientHelloKeys, e = xraytls.ConvertToGoECHKeys(raw)
		if e != nil || len(out.EncryptedClientHelloKeys) == 0 {
			return fail()
		}
		for _, key := range out.EncryptedClientHelloKeys {
			if !validECHKey(key) {
				return fail()
			}
		}
	}
	if path, e := stringValue("masterKeyLog"); e != nil {
		return fail()
	} else if path != "" && path != "none" {
		if !filepath.IsAbs(path) {
			return fail()
		}
		if info, e := os.Lstat(path); e == nil {
			if !info.Mode().IsRegular() || info.Mode().Perm()&0077 != 0 {
				return fail()
			}
		} else if !os.IsNotExist(e) {
			return fail()
		}
		parent, e := os.Stat(filepath.Dir(path))
		if e != nil || !parent.IsDir() {
			return fail()
		}
		out.KeyLogWriter = &privateKeyLog{path: path}
	}
	reject, e := boolValue("rejectUnknownSni")
	if e != nil {
		return fail()
	}
	var issuedMu sync.Mutex
	issued := map[string]*tls.Certificate{}
	out.GetCertificate = func(info *tls.ClientHelloInfo) (*tls.Certificate, error) {
		for _, cert := range serving {
			if cert.pair.Leaf.VerifyHostname(info.ServerName) == nil {
				return cert.withStaple(info.Context())
			}
		}
		if len(issuing) > 0 {
			if info.ServerName == "" || len(info.ServerName) > 253 || strings.ContainsAny(info.ServerName, "\x00\r\n") {
				return nil, errors.New("invalid TLS server name")
			}
			issuedMu.Lock()
			defer issuedMu.Unlock()
			if cached := issued[info.ServerName]; cached != nil && time.Until(cached.Leaf.NotAfter) > 2*time.Minute {
				return cached, nil
			}
			if len(issued) >= 1024 {
				issued = map[string]*tls.Certificate{}
			}
			cert, e := issuing[0].issue(info.ServerName)
			if e == nil {
				issued[info.ServerName] = cert
			}
			return cert, e
		}
		if !reject && len(serving) > 0 {
			return serving[0].withStaple(info.Context())
		}
		return nil, errors.New("unknown TLS server name")
	}
	// Always use the selector, including a single certificate: this enforces
	// SNI rejection, OCSP refresh and issuance on every fresh TLS handshake.
	out.Certificates = nil
	return out, firstCertificate, firstKey, nil
}

func (c *servingCertificate) issue(name string) (*tls.Certificate, error) {
	key, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		return nil, e
	}
	serial, e := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 128))
	if e != nil {
		return nil, e
	}
	until := time.Now().Add(24 * time.Hour)
	if c.pair.Leaf.NotAfter.Before(until) {
		until = c.pair.Leaf.NotAfter
	}
	template := &x509.Certificate{SerialNumber: serial, Subject: pkix.Name{CommonName: name}, NotBefore: time.Now().Add(-time.Minute), NotAfter: until, KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth}, BasicConstraintsValid: true}
	if ip := net.ParseIP(name); ip != nil {
		template.IPAddresses = []net.IP{ip}
	} else {
		template.DNSNames = []string{name}
	}
	der, e := x509.CreateCertificate(rand.Reader, template, c.pair.Leaf, &key.PublicKey, c.pair.PrivateKey)
	if e != nil {
		return nil, e
	}
	leaf, e := x509.ParseCertificate(der)
	if e != nil {
		return nil, e
	}
	out := &tls.Certificate{Certificate: [][]byte{der}, PrivateKey: key, Leaf: leaf}
	if c.buildChain {
		out.Certificate = append(out.Certificate, c.pair.Certificate...)
	}
	return out, nil
}
func (c *servingCertificate) withStaple(ctx context.Context) (*tls.Certificate, error) {
	out := c.pair
	if c.ocspEvery == 0 {
		return &out, nil
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	if len(c.staple) > 0 && time.Now().Before(c.refresh) {
		out.OCSPStaple = append([]byte(nil), c.staple...)
		return &out, nil
	}
	ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
	defer cancel()
	var issuer *x509.Certificate
	var e error
	if len(c.pair.Certificate) > 1 {
		issuer, e = x509.ParseCertificate(c.pair.Certificate[1])
	} else if len(c.pair.Leaf.IssuingCertificateURL) > 0 {
		var raw []byte
		raw, e = tlsFetch(ctx, "GET", c.pair.Leaf.IssuingCertificateURL[0], nil, "")
		if e == nil {
			if block, _ := pem.Decode(raw); block != nil {
				raw = block.Bytes
			}
			issuer, e = x509.ParseCertificate(raw)
		}
	} else {
		e = errors.New("OCSP issuer unavailable")
	}
	if e != nil || issuer == nil || len(c.pair.Leaf.OCSPServer) == 0 {
		return nil, errors.New("OCSP issuer unavailable")
	}
	request, e := ocsp.CreateRequest(c.pair.Leaf, issuer, nil)
	if e != nil {
		return nil, e
	}
	raw, e := tlsFetch(ctx, "POST", c.pair.Leaf.OCSPServer[0], request, "application/ocsp-request")
	if e != nil {
		return nil, e
	}
	response, e := ocsp.ParseResponseForCert(raw, c.pair.Leaf, issuer)
	now := time.Now()
	if e != nil || response.Status != ocsp.Good || response.ThisUpdate.After(now.Add(time.Minute)) || !response.NextUpdate.IsZero() && !response.NextUpdate.After(now) {
		return nil, errors.New("invalid OCSP response")
	}
	c.staple = append([]byte(nil), raw...)
	c.refresh = now.Add(c.ocspEvery)
	if !response.NextUpdate.IsZero() && response.NextUpdate.Before(c.refresh) {
		c.refresh = response.NextUpdate
	}
	out.OCSPStaple = append([]byte(nil), raw...)
	return &out, nil
}
func tlsFetch(ctx context.Context, method, address string, body []byte, contentType string) ([]byte, error) {
	parsed, e := url.Parse(address)
	if e != nil || parsed.Scheme != "http" && parsed.Scheme != "https" || parsed.Host == "" {
		return nil, errors.New("invalid certificate service URL")
	}
	request, e := http.NewRequestWithContext(ctx, method, address, bytes.NewReader(body))
	if e != nil {
		return nil, e
	}
	if contentType != "" {
		request.Header.Set("Content-Type", contentType)
	}
	client := &http.Client{Transport: &http.Transport{Proxy: nil, DisableKeepAlives: true}, CheckRedirect: func(*http.Request, []*http.Request) error { return errors.New("certificate service redirect refused") }}
	response, e := client.Do(request)
	if e != nil {
		return nil, errors.New("certificate service unavailable")
	}
	defer response.Body.Close()
	if response.StatusCode != 200 {
		return nil, errors.New("certificate service refused request")
	}
	raw, e := io.ReadAll(io.LimitReader(response.Body, (1<<20)+1))
	if e != nil || len(raw) > 1<<20 {
		return nil, errors.New("invalid certificate service response")
	}
	return raw, nil
}

type privateKeyLog struct {
	mu     sync.Mutex
	path   string
	file   *os.File
	closed bool
}

func (w *privateKeyLog) Open() error {
	w.mu.Lock()
	defer w.mu.Unlock()
	if w.closed {
		return errors.New("key log closed")
	}
	if w.file != nil {
		return nil
	}
	fd, e := syscall.Open(w.path, syscall.O_WRONLY|syscall.O_CREAT|syscall.O_APPEND|syscall.O_NONBLOCK|syscall.O_NOFOLLOW|syscall.O_CLOEXEC, 0600)
	if e != nil {
		return errors.New("key log unavailable")
	}
	w.file = os.NewFile(uintptr(fd), "private TLS key log")
	info, e := w.file.Stat()
	if e != nil || !info.Mode().IsRegular() || info.Mode().Perm()&0077 != 0 {
		w.file.Close()
		w.file = nil
		return errors.New("key log permissions")
	}
	return nil
}
func (w *privateKeyLog) Write(p []byte) (int, error) {
	w.mu.Lock()
	defer w.mu.Unlock()
	if w.closed || w.file == nil {
		return 0, errors.New("key log unavailable")
	}
	return w.file.Write(p)
}
func (w *privateKeyLog) Close() error {
	w.mu.Lock()
	defer w.mu.Unlock()
	w.closed = true
	if w.file != nil {
		return w.file.Close()
	}
	return nil
}

func validECHKey(key tls.EncryptedClientHelloKey) bool {
	private, e := ecdh.X25519().NewPrivateKey(key.PrivateKey)
	if e != nil {
		return false
	}
	raw := cryptobyte.String(key.Config)
	var version, kem uint16
	var body, pub, ciphers, name, extensions cryptobyte.String
	var id, maxName uint8
	if !raw.ReadUint16(&version) || version != 0xfe0d || !raw.ReadUint16LengthPrefixed(&body) || !raw.Empty() || !body.ReadUint8(&id) || !body.ReadUint16(&kem) || kem != 0x20 || !body.ReadUint16LengthPrefixed(&pub) || !bytes.Equal(pub, private.PublicKey().Bytes()) || !body.ReadUint16LengthPrefixed(&ciphers) || len(ciphers) == 0 || len(ciphers)%4 != 0 || !body.ReadUint8(&maxName) || !body.ReadUint8LengthPrefixed(&name) || len(name) == 0 || len(name) > 253 || strings.ContainsAny(string(name), "\x00\r\n /:") || !body.ReadUint16LengthPrefixed(&extensions) || !body.Empty() {
		return false
	}
	supported := false
	for !ciphers.Empty() {
		var kdf, aead uint16
		if !ciphers.ReadUint16(&kdf) || !ciphers.ReadUint16(&aead) {
			return false
		}
		if kdf == 1 && (aead == 1 || aead == 2 || aead == 3) {
			supported = true
		}
	}
	for !extensions.Empty() {
		var kind uint16
		var value cryptobyte.String
		if !extensions.ReadUint16(&kind) || !extensions.ReadUint16LengthPrefixed(&value) || kind&0x8000 != 0 {
			return false
		}
	}
	return supported
}
