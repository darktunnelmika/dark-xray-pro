func testCertificate(t *testing.T) (string, string, *x509.CertPool) {
	t.Helper()
	key, e := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if e != nil {
		t.Fatal(e)
	}
	tmpl := &x509.Certificate{SerialNumber: big.NewInt(1), Subject: pkix.Name{CommonName: "localhost"}, DNSNames: []string{"localhost"}, IPAddresses: []net.IP{net.IPv4(127, 0, 0, 1)}, NotBefore: time.Now().Add(-time.Minute), NotAfter: time.Now().Add(time.Hour), KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth}}
	der, e := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if e != nil {
		t.Fatal(e)
	}
	private, e := x509.MarshalPKCS8PrivateKey(key)
	if e != nil {
		t.Fatal(e)
	}
	cert := string(pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der}))
	keyPEM := string(pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: private}))
	roots := x509.NewCertPool()
	roots.AppendCertsFromPEM([]byte(cert))
	return cert, keyPEM, roots
}
func inputFor(t *testing.T, controller string, port int, cert, key string, clients []tuic.TuicClientSettings) darkproadapter.Input {
	t.Helper()
	cfg := config{Listen: net.JoinHostPort("127.0.0.1", strconv.Itoa(port)), Server: tuic.TuicServerSettings{CongestionControl: controller, MaxIdleTime: 30, AuthenticationTimeout: 1, MaxUdpRelayPacketSize: 1500, ALPN: []string{"h3"}}, Clients: clients, TLS: map[string]json.RawMessage{}}
	cfg.TLS["certificates"], _ = json.Marshal([]map[string]any{{"certificate": strings.Split(cert, "\n"), "key": strings.Split(key, "\n")}})
	cfg.TLS["rejectUnknownSni"] = json.RawMessage("true")
	raw, _ := json.Marshal(cfg)
	in := darkproadapter.Input{Version: 1, Engine: "tuic", Config: raw}
	for i, c := range clients {
		role := "client"
		if i == 1 {
			role = "health"
		}
		in.Principals = append(in.Principals, darkproadapter.Principal{ID: c.Email, Engine: "tuic", Role: role, Counter: "user>>>" + c.Email + ">>>traffic", InboundID: 1})
	}
	return in
}
func managed(t *testing.T, in darkproadapter.Input) *darkproadapter.Worker {
	t.Helper()
	p, e := prepare(in)
	if e != nil {
		t.Fatal(e)
	}
	w := dt.Worker(t, in, p)
	var cfg config
	_ = json.Unmarshal(in.Config, &cfg)
	dt.AssertUDPClosed(t, cfg.Listen)
	dt.Command(t, w, "activate", dt.ActivateNonce)
	return w
}
func dialTUIC(t *testing.T, address string, roots *x509.CertPool, user tuic.TuicClientSettings) *quic.Conn {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	defer cancel()
	c, e := quic.DialAddr(ctx, address, &tls.Config{RootCAs: roots, ServerName: "localhost", NextProtos: []string{"h3"}, MinVersion: tls.VersionTLS13}, &quic.Config{EnableDatagrams: true})
	if e != nil {
		t.Fatal(e)
	}
	id, e := hex.DecodeString(strings.ReplaceAll(user.UUID, "-", ""))
	if e != nil {
		t.Fatal(e)
	}
	state := c.ConnectionState().TLS
	token, e := state.ExportKeyingMaterial(string(id), []byte(user.Password), 32)
	if e != nil {
		t.Fatal(e)
	}
	s, e := c.OpenUniStreamSync(ctx)
	if e != nil {
		t.Fatal(e)
	}
	raw := append([]byte{tuic.ProtocolVersion, tuic.CmdAuthenticate}, id...)
	raw = append(raw, token...)
	if _, e = s.Write(raw); e != nil {
		t.Fatal(e)
	}
	_ = s.Close()
	t.Cleanup(func() { _ = c.CloseWithError(0, "") })
	return c
}
func address(t *testing.T, target string) *tuic.Address {
	t.Helper()
	host, port, e := net.SplitHostPort(target)
	if e != nil {
		t.Fatal(e)
	}
	p, e := strconv.Atoi(port)
	if e != nil {
		t.Fatal(e)
	}
	return &tuic.Address{Type: tuic.AddrTypeIPv4, IP: net.ParseIP(host), Port: uint16(p)}
}
func tcpTUIC(t *testing.T, c *quic.Conn, target string, payload []byte) *quic.Stream {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	defer cancel()
	s, e := c.OpenStreamSync(ctx)
	if e != nil {
		t.Fatal(e)
	}
	_ = s.SetDeadline(time.Now().Add(5 * time.Second))
	var b bytes.Buffer
	b.Write([]byte{tuic.ProtocolVersion, tuic.CmdConnect})
	if e = tuic.WriteAddress(&b, address(t, target)); e != nil {
		t.Fatal(e)
	}
	b.Write(payload)
	if _, e = s.Write(b.Bytes()); e != nil {
		t.Fatal(e)
	}
	out := make([]byte, len(payload))
	if _, e = io.ReadFull(s, out); e != nil {
		t.Fatal(e)
	}
	if !bytes.Equal(payload, out) {
		t.Fatal("TUIC TCP payload mismatch")
	}
	return s
}
func udpTUIC(t *testing.T, c *quic.Conn, target string, payload []byte, stream bool) {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	defer cancel()
	var b bytes.Buffer
	assoc := uint16(42)
	if stream {
		assoc = 43
	}
	if e := tuic.WritePacket(&b, assoc, 1, 1, 0, address(t, target), payload); e != nil {
		t.Fatal(e)
	}
	var reader io.Reader
	if stream {
		s, e := c.OpenUniStreamSync(ctx)
		if e != nil {
			t.Fatal(e)
		}
		if _, e = s.Write(b.Bytes()); e != nil {
			t.Fatal(e)
		}
		_ = s.Close()
		reply, e := c.AcceptUniStream(ctx)
		if e != nil {
			t.Fatal(e)
		}
		_ = reply.SetReadDeadline(time.Now().Add(5 * time.Second))
		reader = reply
	} else {
		if e := c.SendDatagram(b.Bytes()); e != nil {
			t.Fatal(e)
		}
		raw, e := c.ReceiveDatagram(ctx)
		if e != nil {
			t.Fatal(e)
		}
		reader = bytes.NewReader(raw)
	}
	prefix := make([]byte, 2)
	if _, e := io.ReadFull(reader, prefix); e != nil || !bytes.Equal(prefix, []byte{tuic.ProtocolVersion, tuic.CmdPacket}) {
		t.Fatal("invalid TUIC UDP response")
	}
	hdr, e := tuic.ReadPacketHeader(reader)
	if e != nil || hdr.FragTotal != 1 || hdr.Size != uint16(len(payload)) {
		t.Fatal("TUIC UDP header mismatch")
	}
	out := make([]byte, len(payload))
	if _, e = io.ReadFull(reader, out); e != nil || !bytes.Equal(out, payload) {
		t.Fatalf("TUIC UDP payload mismatch: %v", e)
	}
}
func TestManagedTUICActualTCPUDPAccountingAndRevoke(t *testing.T) {
	for _, controller := range []string{"bbr", "cubic", "new_reno"} {
		t.Run(controller, func(t *testing.T) {
			cert, key, roots := testCertificate(t)
			port := dt.Port(t)
			addr := net.JoinHostPort("127.0.0.1", strconv.Itoa(port))
			alice := tuic.TuicClientSettings{UUID: "11111111-1111-4111-8111-111111111111", Password: "test-secret-one", Email: "alice"}
			probe := tuic.TuicClientSettings{UUID: "22222222-2222-4222-8222-222222222222", Password: "test-secret-two", Email: "probe"}
			in := inputFor(t, controller, port, cert, key, []tuic.TuicClientSettings{alice, probe})
			w := managed(t, in)
			tcpTarget, udpTarget := dt.Echo(t, "127.0.0.1")
			c := dialTUIC(t, addr, roots, alice)
			tcpPayload := []byte("actual TUIC payload " + controller)
			s := tcpTUIC(t, c, tcpTarget, tcpPayload)
			udpPayload := []byte("actual TUIC datagram")
			udpTUIC(t, c, udpTarget, udpPayload, false)
			udpTUIC(t, c, udpTarget, udpPayload, true)
			health := dialTUIC(t, addr, roots, probe)
			healthPayload := []byte("nonbillable health")
			hs := tcpTUIC(t, health, tcpTarget, healthPayload)
			_ = hs.Close()
			_ = health.CloseWithError(0, "")
			want := []darkproadapter.Counter{{ID: "alice", Upload: uint64(len(tcpPayload) + 2*len(udpPayload)), Download: uint64(len(tcpPayload) + 2*len(udpPayload))}, {ID: "probe", Upload: uint64(len(healthPayload)), Download: uint64(len(healthPayload))}}
			dt.Counters(t, w, want)
			wrong := alice
			wrong.Password = "wrong-secret"
			denied := dialTUIC(t, addr, roots, wrong)
			select {
			case <-denied.Context().Done():
			case <-time.After(3 * time.Second):
				t.Fatal("wrong TUIC credentials admitted")
			}
			dt.Counters(t, w, want)
			join := dt.InFlight(t, s)
			final := dt.Final(t, w)
			join()
			_ = s.SetReadDeadline(time.Now().Add(time.Second))
			if _, e := s.Read(make([]byte, 1)); e == nil {
				t.Fatal("TUIC stream remained live after final")
			}
			dt.Counters(t, w, final.Counters)
			if controller == "cubic" {
				replacement := managed(t, inputFor(t, controller, port, cert, key, []tuic.TuicClientSettings{probe}))
				revoked := dialTUIC(t, addr, roots, alice)
				select {
				case <-revoked.Context().Done():
				case <-time.After(3 * time.Second):
					t.Fatal("revoked TUIC identity accepted")
				}
				fresh := dialTUIC(t, addr, roots, probe)
				_ = tcpTUIC(t, fresh, tcpTarget, healthPayload).Close()
				dt.Counters(t, replacement, []darkproadapter.Counter{{ID: "probe", Upload: uint64(len(healthPayload)), Download: uint64(len(healthPayload))}})
			}
		})
	}
}
func TestTUICPreparationRejectsUnavailableMaterialAndTLS12(t *testing.T) {
	cert, key, _ := testCertificate(t)
	user := tuic.TuicClientSettings{UUID: "11111111-1111-4111-8111-111111111111", Password: "secret", Email: "alice"}
	in := inputFor(t, "cubic", dt.Port(t), cert, key, []tuic.TuicClientSettings{user})
	var cfg config
	_ = json.Unmarshal(in.Config, &cfg)
	cfg.TLS["maxVersion"] = json.RawMessage(`"1.2"`)
	in.Config, _ = json.Marshal(cfg)
	if _, e := prepare(in); e == nil {
		t.Fatal("TUIC accepted TLS 1.2 maximum")
	}
	delete(cfg.TLS, "maxVersion")
	cfg.TLS["certificates"], _ = json.Marshal([]map[string]any{{"certificateFile": "/definitely-missing/certificate.pem", "keyFile": "/definitely-missing/key.pem"}})
	in.Config, _ = json.Marshal(cfg)
	if _, e := prepare(in); e == nil {
		t.Fatal("TUIC accepted missing TLS material")
	}
}
